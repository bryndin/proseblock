{{- /* share.html — Config-driven share menu popover. Reads share platforms from site.Params.share.platforms, falling back to sensible defaults if none are configured. @context {page} . The current page context. */ -}} {{- $page := . -}} {{- /* ─── Gate: bail if sharing is disabled ─── */ -}} {{- with $shareConfig := site.Params.share -}} {{- if $shareConfig.enable -}} {{- /* ─── Resolve page data for URL templates ─── */ -}} {{- $title := $page.Title | safeURL -}} {{- $url := $page.Permalink | safeURL -}} {{- /* ─── Use configured platforms ─── */ -}} {{- $platforms := $shareConfig.platforms -}} {{- /* ─── Only render if we have at least one platform ─── */ -}} {{- if gt (len $platforms) 0 -}}
{{- /* Trigger button */ -}} {{- /* Popover menu */ -}}
{{- range $platform := $platforms -}} {{- $href := $platform.url -}} {{- $href = replace $href "{title}" $title -}} {{- $href = replace $href "{url}" $url -}} {{- $icon := $platform.icon | default "share" -}} {{- $isEmail := hasPrefix $href "mailto:" -}} {{- partial (printf "svg/%s.html" $icon) -}} {{ $platform.name }} {{- end -}}
{{- end -}} {{- end -}} {{- end -}}

{{- /* CSS Pipeline: Automatic Layer Encapsulation Architecture: reset, tokens, base, layout, components, utilities Context: . (The Page) */ -}} {{- $page := . -}} {{- $syntaxLibHint := "github-dark.css" -}} {{- $layerNames := slice "reset" "tokens" "vendor" "base" "prose" "layout" "components" "utilities" -}} {{- /* Syntax-highlighting lib must be loaded before the components */ -}} {{- $layerMap := dict "reset" (slice "css/reset.css") "tokens" (slice "css/tokens.css") "vendor" (slice "vendor/**/*.css") "base" (slice "css/base.css" "css/base/*.css") "prose" (slice "css/prose/*.css") "layout" (slice "css/layout/*.css") "components" (slice "css/components/*.css" "css/components/widgets/*.css") "utilities" (slice "css/utilities.css") -}} {{- /* 1. Build the entire CSS payload as a single string */ -}} {{- $fullCSS := printf "@layer %s;\n\n" (delimit $layerNames ", ") -}} {{- range $name := $layerNames -}} {{- $currentLayerFiles := slice -}} {{- /* Gather static files */ -}} {{- range $pattern := index $layerMap $name -}} {{- range $match := resources.Match $pattern -}} {{- $currentLayerFiles = $currentLayerFiles | append $match -}} {{- end -}} {{- end -}} {{- /* Gather dynamic components */ -}} {{- if eq $name "components" -}} {{- with $page.Store.Get "bundle-css" -}} {{- range $resource := . -}} {{- $currentLayerFiles = $currentLayerFiles | append $resource -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Extract and wrap contents natively */ -}} {{- if gt (len $currentLayerFiles) 0 -}} {{- $layerContent := slice -}} {{- range $res := $currentLayerFiles -}} {{- /* Reading .Content directly grabs fresh text on file save */ -}} {{- if in $res.Name $syntaxLibHint -}} {{- $layerContent = $layerContent | append (printf "/* %s */\n[data-theme=\"dark\"] {\n%s\n}" $res.Name $res.Content) -}} {{- else -}} {{- $layerContent = $layerContent | append (printf "/* %s */\n%s" $res.Name $res.Content) -}} {{- end -}} {{- end -}} {{- /* Append this layer to the global string */ -}} {{- $fullCSS = printf "%s@layer %s {\n%s\n}\n\n" $fullCSS $name (delimit $layerContent "\n") -}} {{- end -}} {{- end -}} {{- /* 2. Bypass Hugo's internal cache trap */ -}} {{- /* By hashing the intermediate string, we force Hugo to recognize the new content. */ -}} {{- $contentHash := md5 $fullCSS -}} {{- $rawResource := $fullCSS | resources.FromString (printf "css/internal/bundle-raw-%s.css" $contentHash) -}} {{- /* 3. Final Execution */ -}} {{- /* CRITICAL FIX: The target path here MUST be statically named in development! */ -}} {{- $targetPath := "css/bundle.css" -}} {{- $css := $rawResource | resources.ExecuteAsTemplate $targetPath $page -}} {{- /* 4. Output Final Link Tag */ -}} {{- if hugo.IsDevelopment -}} {{- /* Live Reload will correctly match this static href and hot-swap the CSS */ -}} {{- else -}} {{- /* Fingerprinting handles production caching natively */ -}} {{- $css = $css | minify | fingerprint -}} {{- end -}}
